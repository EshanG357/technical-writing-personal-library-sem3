\begin{frame}{What is Technical Writing?}
  Technical writing communicates specialised information clearly.
  \begin{block}{Main objective}
    Make technical information understandable and usable.
  \end{block}
\end{frame}

\begin{frame}{A Structured Workflow}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Planning}
        Identify the audience and purpose.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Writing}
        Organise information and use examples.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Incremental Content}
  \begin{itemize}
    \item<1-> Define the topic.
    \item<2-> Explain the concepts.
    \item<3-> Demonstrate with an example.
  \end{itemize}
\end{frame}
